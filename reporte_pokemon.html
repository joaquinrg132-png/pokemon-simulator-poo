📋 Conceptos de POO Implementados

🏗️ Constructor __init__

Inicializa todos los atributos de instancia (nombre, tipo, nivel, ataque, defensa, salud) usando self. Incrementa el contador de clase y muestra mensaje de captura.

💣 Destructor __del__

Se ejecuta automáticamente al eliminar la última referencia al objeto. Decrementa Pokemon._total y muestra el mensaje de liberación.

🔄 Sobrecarga con *args

entrenar(*args) acepta 0, 1, 2 ó 3+ argumentos, simulando sobrecarga de métodos: automático, solo ataque, ataque+defensa, o los tres stats.

🏷️ Método de Clase

@classmethod total_pokemons(cls) accede a la variable de clase _total sin necesitar una instancia. Cuenta Pokémon activos.

🌍 Variables Globales / Locales

pokemons_capturados es global en main.py; variables como daño, bonus_ataque, porcentaje son locales dentro de sus métodos.

↔️ Paso de Parámetros

Tipos primitivos (str, int) se pasan por valor en el constructor. El objeto objetivo en atacar() se pasa por referencia, modificando su salud directamente.

"""
=============================================================
  Simulador de Entrenamiento Pokémon
  Módulo: clases/pokemon.py
=============================================================
"""

class Pokemon:
    """Representa a un Pokémon con sus estadísticas y comportamientos."""

    # ── Variable de clase (compartida entre todas las instancias) ─────
    _total: int = 0  # Cuenta cuántos Pokémon existen actualmente

    # ── CONSTRUCTOR ───────────────────────────────────────────────────
    def __init__(self, nombre: str, tipo: str,
                 ataque: int, defensa: int, salud: int) -> None:
        """Inicializa un nuevo Pokémon. 'nivel' siempre inicia en 1."""
        # Atributos de instancia accedidos con self
        self.nombre    = nombre       # Paso por valor (str inmutable)
        self.tipo      = tipo
        self.nivel     = 1            # Nivel inicial siempre es 1
        self.ataque    = ataque
        self.defensa   = defensa
        self.salud     = salud
        self.salud_max = salud        # Salud máxima para referencia
        Pokemon._total += 1          # Actualizar contador de clase
        print(f"\n  ✨ ¡{self.nombre} ({self.tipo}) ha sido capturado! "
              f"[Nivel {self.nivel}]")

    # ── DESTRUCTOR ────────────────────────────────────────────────────
    def __del__(self) -> None:
        """Se ejecuta al eliminar el objeto. Muestra mensaje de despedida."""
        try:
            Pokemon._total -= 1
            print(f"\n  🕊️  {self.nombre} ha sido liberado ✨")
        except Exception:
            pass

    # ── MÉTODO DE INSTANCIA: entrenar() con sobrecarga *args ──────────
    def entrenar(self, *args) -> None:
        """
        Entrena al Pokémon. Sobrecarga con *args:
          0 args → mejora automática escalada por nivel
          1 arg  → solo ataque
          2 args → ataque + defensa
          3 args → ataque + defensa + salud
        """
        # Variables locales (ámbito de función)
        bonus_ataque  = 0
        bonus_defensa = 0
        bonus_salud   = 0

        if len(args) == 0:
            bonus_ataque  = 5 + self.nivel * 2
            bonus_defensa = 3 + self.nivel
            bonus_salud   = 10 + self.nivel * 3
        elif len(args) == 1:
            bonus_ataque = int(args[0])
        elif len(args) == 2:
            bonus_ataque  = int(args[0])
            bonus_defensa = int(args[1])
        else:
            bonus_ataque  = int(args[0])
            bonus_defensa = int(args[1])
            bonus_salud   = int(args[2])

        self.nivel    += 1             # Subir nivel
        self.ataque   += bonus_ataque  # Modificación por referencia del objeto
        self.defensa  += bonus_defensa
        self.salud    += bonus_salud
        self.salud_max = self.salud

        print(f"\n  💪 ¡{self.nombre} ha entrenado duro!")
        print(f"     Nivel     : {self.nivel - 1} → {self.nivel}")
        if bonus_ataque  > 0: print(f"     Ataque    : +{bonus_ataque}")
        if bonus_defensa > 0: print(f"     Defensa   : +{bonus_defensa}")
        if bonus_salud   > 0: print(f"     Salud     : +{bonus_salud}")

    # ── MÉTODO DE INSTANCIA: mostrar_info() ───────────────────────────
    def mostrar_info(self) -> None:
        """Muestra todos los datos del Pokémon accediendo con self."""
        barra = self._barra_salud()  # variable local
        print(f"""
  ╔══════════════════════════════════════╗
  ║  🐾  {self.nombre:<33}║
  ╠══════════════════════════════════════╣
  ║  Tipo     : {self.tipo:<27}║
  ║  Nivel    : {self.nivel:<27}║
  ║  Ataque   : {self.ataque:<27}║
  ║  Defensa  : {self.defensa:<27}║
  ║  Salud    : {self.salud}/{self.salud_max:<21}║
  ║  HP       : {barra:<27}║
  ╚══════════════════════════════════════╝""")

    # ── MÉTODO DE INSTANCIA: atacar(objetivo) ─────────────────────────
    def atacar(self, objetivo: "Pokemon") -> None:
        """
        Ataca a otro Pokémon (paso por referencia).
        Daño = max(1, ataque - defensa_objetivo // 2)
        """
        import random  # import local
        # Variables locales
        critico = random.randint(1, 5) == 1
        daño    = max(1, self.ataque - objetivo.defensa // 2)
        if critico: daño = int(daño * 1.5)
        # Modificar salud del objetivo (referencia al objeto)
        objetivo.salud = max(0, objetivo.salud - daño)
        tipo_ataque = "¡CRÍTICO! 💥" if critico else "⚔️"
        print(f"\n  {tipo_ataque}  {self.nombre} ataca a {objetivo.nombre}!")
        print(f"     Daño causado  : {daño}")
        print(f"     Salud restante: {objetivo.salud}/{objetivo.salud_max}")
        if objetivo.salud == 0:
            print(f"     💀 ¡{objetivo.nombre} ha sido derrotado!")

    # ── MÉTODO DE CLASE ───────────────────────────────────────────────
    @classmethod
    def total_pokemons(cls) -> None:
        """Muestra cuántos Pokémon existen (variable de clase cls._total)."""
        print(f"\n  🧮 Pokémon activos en el sistema: {cls._total}")

    def poder_total(self) -> int:
        """Calcula (ataque + defensa + salud) * nivel."""
        poder = (self.ataque + self.defensa + self.salud) * self.nivel  # var local
        return poder

    def _barra_salud(self) -> str:
        """Genera barra de salud visual de 10 caracteres."""
        if self.salud_max == 0: return "░░░░░░░░░░"
        porcentaje = self.salud / self.salud_max  # variable local
        llenas = int(porcentaje * 10)
        return "█" * llenas + "░" * (10 - llenas)

    def __str__(self) -> str:
        return (f"{self.nombre} [{self.tipo}] Lv.{self.nivel} | "
                f"ATK:{self.ataque} DEF:{self.defensa} HP:{self.salud}")
